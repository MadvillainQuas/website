# The public section is meant to be found. The portal, the scorer and the
# admin console are not — they are behind Supabase auth regardless, but there
# is no reason to spend a crawler's budget discovering that.
User-agent: *
Allow: /epinoia/
Disallow: /epinoia/app/
Disallow: /epinoia/score/
Disallow: /epinoia/admin/
Disallow: /epinoia/devfeed.html
# An invitation to a private league carries its token in the address. The page
# is noindex anyway, but a crawler that never fetches it cannot put the token
# in a log, a referrer or an index in the first place.
Disallow: /epinoia/invite/
